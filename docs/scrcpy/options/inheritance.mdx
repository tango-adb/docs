# Inheritance

{/* Source: libraries/scrcpy/src/base/options.ts | Commit: 6bb9151c3d5bc6cf1e94d7684e99e7cfa3c827b1 */}

Each `ScrcpyOptionsX_YY` class implements the shared `ScrcpyOptions<T>` interface. New server versions can change how a feature works, so each class provides the behavior needed to parse and serialize its version of the protocol.

The interface includes the normalized option values and the operations used by the client:

```ts
import type { MaybePromiseLike } from "@yume-chan/async";
import type {
    Consumable,
    ReadableStream,
    TransformStream,
    WritableStreamDefaultWriter,
} from "@yume-chan/stream-extra";

// import type {
//     ScrcpyAudioStreamMetadata,
//     ScrcpyBackOrScreenOnControlMessage,
//     ScrcpyControlMessageType,
//     ScrcpyDeviceMessageParser,
//     ScrcpyDisplay,
//     ScrcpyInjectTouchControlMessage,
//     ScrcpyOptions,
//     ScrcpyScrollController,
//     ScrcpySetClipboardControlMessage,
//     ScrcpySetDisplayPowerControlMessage,
//     ScrcpyUHidCreateControlMessage,
//     ScrcpyUHidManager,
//     ScrcpyVideoStream,
//     ScrcpyVideoStreamPacket,
// } from "@yume-chan/scrcpy";

interface ScrcpyOptions<T extends object> {
    get controlMessageTypes(): Partial<Record<ScrcpyControlMessageType, number>>;

    value: T;

    createDeviceMessageParser: (() => ScrcpyDeviceMessageParser) | undefined;

    serialize(): string[];
    setListDisplays(): void;
    parseDisplay(line: string): ScrcpyDisplay | undefined;
    parseVideoStreamMetadata(
        stream: ReadableStream<Uint8Array>,
    ): MaybePromiseLike<ScrcpyVideoStream>;
    parseAudioStreamMetadata?(
        stream: ReadableStream<Uint8Array>,
    ): MaybePromiseLike<ScrcpyAudioStreamMetadata>;
    createMediaStreamTransformer(): TransformStream<
        Uint8Array,
        ScrcpyVideoStreamPacket
    >;
    serializeInjectTouchControlMessage(
        message: ScrcpyInjectTouchControlMessage,
    ): Uint8Array;
    serializeBackOrScreenOnControlMessage(
        message: ScrcpyBackOrScreenOnControlMessage,
    ): Uint8Array | undefined;
    serializeSetClipboardControlMessage(
        message: ScrcpySetClipboardControlMessage,
        deviceMessageParser: ScrcpyDeviceMessageParser,
    ): Uint8Array | [Uint8Array, Promise<void>];
    serializeSetDisplayPowerControlMessage(
        message: ScrcpySetDisplayPowerControlMessage,
    ): Uint8Array;
    createScrollController(): ScrcpyScrollController;
    serializeUHidCreateControlMessage?(
        message: ScrcpyUHidCreateControlMessage,
    ): Uint8Array;
    createUHidManager?: (
        writer: WritableStreamDefaultWriter<Consumable<Uint8Array>>,
        deviceMessageParser: ScrcpyDeviceMessageParser,
    ) => ScrcpyUHidManager;
}
```

`createDeviceMessageParser` creates the parser for messages received from the device. It is `undefined` when the selected options don't support device messages. `createUHidManager` is available only for options that support UHID and have control enabled; see [UHID manager](../uhid/manager.mdx).
